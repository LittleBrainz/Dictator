add_library(Dictator.Native SHARED src/dictator_native.cpp)
target_include_directories(Dictator.Native PUBLIC include)
target_compile_definitions(Dictator.Native PRIVATE DICTATOR_NATIVE_EXPORTS)
set_target_properties(Dictator.Native PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(WIN32)
  target_compile_definitions(Dictator.Native PRIVATE UNICODE _UNICODE)
  target_sources(Dictator.Native PRIVATE src/resident_host.cpp src/target_probe.cpp src/widget_painter.cpp "${CMAKE_BINARY_DIR}/Dictator.icon.rc")
  target_link_libraries(Dictator.Native PRIVATE user32 shell32 gdi32 advapi32 ole32 oleaut32 uiautomationcore shcore gdiplus)
endif()
if(MSVC)
  target_compile_options(Dictator.Native PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(Dictator.Native PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
